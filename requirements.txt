# Chitrak locomotion — pinned to the exact set verified working together.
# Python 3.11 ONLY (Isaac Sim ships no wheels for 3.12+).
#
#   pip install -r requirements.txt --extra-index-url https://pypi.nvidia.com
#
# Isaac Lab itself is NOT here — it is a git clone + editable install, see README §2.

# --- simulator -------------------------------------------------------------
# Pulls in ~40 isaacsim-* subpackages and pins torch to 2.7.0+cu128.
# The [extscache] extra pre-bakes the Omniverse extension cache; without it the
# first launch spends ~10 minutes downloading extensions.
isaacsim[all,extscache]==5.1.0

# --- RL --------------------------------------------------------------------
# 3.0.1 is the version the local Isaac Lab patch in README §2.3 matches.
rsl-rl-lib==3.0.1
gymnasium==1.2.1

# --- numerics --------------------------------------------------------------
# numpy MUST stay <2 — isaacsim 5.1.0 is built against the 1.x ABI.
numpy==1.26.0
scipy==1.15.3

# --- offline rendering / analysis ------------------------------------------
# MuJoCo is used ONLY for forward kinematics and for rendering logged
# trajectories to mp4 (Isaac Sim's RTX renderer hangs on GPUs without RT cores).
# Never use it for quantitative torque/contact numbers — see README §7.
mujoco==3.10.0
imageio==2.37.0
imageio-ffmpeg

# --- logging ---------------------------------------------------------------
# 0.28.0 is the version verified on the source machine (MIGRATION.md §3).
# 0.26.x has a regression in the run-state handshake — the "state: crashed"
# symptom (README §4) is worse on 0.26.x than on 0.28.x.
# NOTE: stable-baselines3 (pulled transitively by isaaclab_tasks) warns about
# needing torch >= 2.8; this is harmless — RSL-RL does not use sb3.
wandb==0.28.0

# torch is installed transitively by isaacsim-core as 2.7.0+cu128.
# Do NOT upgrade it; it is an ABI pin, not a preference.
